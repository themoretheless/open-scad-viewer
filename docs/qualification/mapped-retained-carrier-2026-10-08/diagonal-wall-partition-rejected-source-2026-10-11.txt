// Partition D x D into L x L, L x R and R x R modulo endpoint
// exchange on the same immutable face. This skips intermediate half-products,
// not geometric candidates; unresolved children remain explicit queue entries.
fn diagonal_children(cell:&ExpressionWallCell,axis:usize,midpoint:f64)->Option<[ExpressionWallCell;3]> {
    if axis>=2 || cell.faces[0]!=cell.faces[1]
        || cell.domains[0].map(|d|d.map(f64::to_bits))!=cell.domains[1].map(|d|d.map(f64::to_bits))
        || !midpoint.is_finite() || midpoint<=cell.domains[0][axis][0] || midpoint>=cell.domains[0][axis][1] {
        return None;
    }
    let mut left=cell.domains[0];let mut right=left;
    left[axis][1]=midpoint;right[axis][0]=midpoint;
    Some([[left,left],[left,right],[right,right]].map(|domains|
        ExpressionWallCell{faces:cell.faces,domains,pending_reason:None}))
}

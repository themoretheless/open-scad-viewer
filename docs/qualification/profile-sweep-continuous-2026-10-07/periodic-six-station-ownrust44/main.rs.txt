use value_codec::{Value,json};
fn main(){
 let args:Vec<_>=std::env::args().collect();
 let mut patches:Value=value_codec::from_str(&std::fs::read_to_string(&args[1]).unwrap()).unwrap();
 for p in patches.as_array_mut().unwrap(){
  let degree=p["degreeV"].as_u64().unwrap() as usize;
  let knots=p["knotsV"].as_array().unwrap();
  let a=knots[degree].as_f64().unwrap();
  let b=knots[knots.len()-degree-1].as_f64().unwrap();
  assert!(a.is_finite() && b.is_finite() && a<b);
  for knot in p["knotsV"].as_array_mut().unwrap(){
   let t=knot.as_f64().unwrap();
   *knot=json!(if t==a {0.}else if t==b {1.}else{(t-a)/(b-a)});
  }
  p["periodicV"]=json!(false);
 }
 std::fs::write(&args[2],value_codec::to_string(&patches).unwrap()).unwrap();
}
